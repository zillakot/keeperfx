cmake_minimum_required(VERSION 3.20)
project(lens_tests C CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
set(KFX_GPU_LIBRARY "" CACHE FILEPATH "Optional real wgpu static library")
add_executable(lens_test lens_test.cpp "${KFX_ROOT}/src/kfx/lense/WgpuLens.cpp"
    "${KFX_ROOT}/src/kfx/renderer/GpolyCapture.c")
target_include_directories(lens_test PRIVATE "${KFX_ROOT}/src")
target_compile_options(lens_test PRIVATE -Wall -Wextra -Werror)
if(KFX_GPU_LIBRARY)
    target_sources(lens_test PRIVATE "${KFX_ROOT}/src/kfx/renderer/WgpuTerrainBridge.cpp")
    target_compile_definitions(lens_test PRIVATE KFX_RUST_PRESENTER=1)
    target_link_libraries(lens_test PRIVATE "${KFX_GPU_LIBRARY}")
    if(APPLE)
        target_link_libraries(lens_test PRIVATE "-framework Metal" "-framework QuartzCore" "-framework Foundation" "-framework AppKit" "-framework IOKit" "-framework CoreGraphics")
    else()
        target_link_libraries(lens_test PRIVATE dl pthread m)
    endif()
else()
    target_compile_options(lens_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(lens_test PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME lens COMMAND lens_test "${CMAKE_CURRENT_BINARY_DIR}/lenses.bin")
