cmake_minimum_required(VERSION 3.20)
project(gpoly_capture_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
set(KFX_SDL3_INCLUDE_DIR "" CACHE PATH "SDL3 include directory for header-only fixture compilation")
if(KFX_SDL3_INCLUDE_DIR)
    set(SDL3_INCLUDE_DIRS "${KFX_SDL3_INCLUDE_DIR}")
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL3 REQUIRED sdl3)
endif()
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ver_defs.h" "#define VER_STRING \"gpoly-test\"\n")
add_executable(gpoly_capture_test capture_test.c "${KFX_ROOT}/src/kfx/renderer/GpolyCapture.c")
target_include_directories(gpoly_capture_test PRIVATE "${KFX_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}" ${SDL3_INCLUDE_DIRS})
target_compile_definitions(gpoly_capture_test PRIVATE BFDEBUG_LEVEL=0)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(gpoly_capture_test PRIVATE -fwrapv -Wall -Wextra -Wno-unknown-pragmas)
endif()
option(KFX_GPOLY_ASAN "Enable address sanitizer" OFF)
if(KFX_GPOLY_ASAN)
    target_compile_options(gpoly_capture_test PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(gpoly_capture_test PRIVATE -fsanitize=address)
endif()
add_test(NAME gpoly_capture COMMAND gpoly_capture_test)
