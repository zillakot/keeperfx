cmake_minimum_required(VERSION 3.20)
project(cursor_tests C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
set(KFX_GPU_LIBRARY "" CACHE FILEPATH "Rust live-surface static library")
if(NOT KFX_GPU_LIBRARY)
    message(FATAL_ERROR "KFX_GPU_LIBRARY is required")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ver_defs.h" "#define VER_STRING \"cursor-test\"\n")
add_executable(cursor_test cursor_test.cpp native.c
    "${KFX_ROOT}/src/bflib_mspointer.cpp"
    "${KFX_ROOT}/src/bflib_vidsurface.c"
    "${KFX_ROOT}/src/kfx/renderer/WgpuCursor.cpp"
    "${KFX_ROOT}/src/kfx/renderer/software/bflib_vidraw_spr_norm.c")
set_source_files_properties(native.c PROPERTIES COMPILE_OPTIONS "-fno-sanitize=alignment")
target_include_directories(cursor_test PRIVATE "${KFX_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(cursor_test PRIVATE KFX_RUST_PRESENTER BFDEBUG_LEVEL=0)
target_compile_options(cursor_test PRIVATE -fwrapv -ffunction-sections -fdata-sections -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(cursor_test PRIVATE -fsanitize=address,undefined)
target_link_libraries(cursor_test PRIVATE "${KFX_GPU_LIBRARY}" PkgConfig::SDL3)
if(APPLE)
    target_link_options(cursor_test PRIVATE -Wl,-dead_strip)
    target_link_libraries(cursor_test PRIVATE "-framework Metal" "-framework QuartzCore" "-framework Foundation" "-framework AppKit" "-framework IOKit" "-framework CoreGraphics")
else()
    target_link_options(cursor_test PRIVATE -Wl,--gc-sections)
    target_link_libraries(cursor_test PRIVATE dl pthread m)
endif()
add_test(NAME cursor COMMAND cursor_test)
