cmake_minimum_required(VERSION 3.20)
project(performance_scope_tests CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
file(READ "${KFX_ROOT}/src/performance_capture.cpp" SOURCE)
string(FIND "${SOURCE}" "void performance_begin(" START)
string(FIND "${SOURCE}" "void performance_renderer_info(" STOP)
if(START LESS 0 OR STOP LESS START)
    message(FATAL_ERROR "Performance scope functions not found")
endif()
math(EXPR LENGTH "${STOP} - ${START}")
string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} FUNCTIONS)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/timing.inc" "${FUNCTIONS}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${KFX_ROOT}/src/performance_capture.cpp")
add_executable(performance_scope_test scope_test.cpp)
target_include_directories(performance_scope_test PRIVATE "${KFX_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(performance_scope_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME performance_scopes COMMAND performance_scope_test)

string(FIND "${SOURCE}" "constexpr int DrawingCounterCount" COUNTERS_START)
string(FIND "${SOURCE}" "struct Profile {" COUNTERS_STOP)
math(EXPR COUNTERS_LENGTH "${COUNTERS_STOP} - ${COUNTERS_START}")
string(SUBSTRING "${SOURCE}" ${COUNTERS_START} ${COUNTERS_LENGTH} COUNTERS)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/counters.inc" "${COUNTERS}")
add_executable(performance_counter_test counter_test.cpp)
target_include_directories(performance_counter_test PRIVATE "${KFX_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(performance_counter_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME performance_counter_layout COMMAND performance_counter_test)
