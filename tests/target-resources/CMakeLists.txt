cmake_minimum_required(VERSION 3.20)
project(target_resource_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
set(KFX_GPU_LIBRARY "" CACHE FILEPATH "Real wgpu static library with live-surface enabled")
if(NOT KFX_GPU_LIBRARY)
    message(FATAL_ERROR "KFX_GPU_LIBRARY is required")
endif()
add_executable(target_resource_test target_resource_test.c)
target_include_directories(target_resource_test PRIVATE "${KFX_ROOT}/src")
target_compile_options(target_resource_test PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(target_resource_test PRIVATE -fsanitize=address,undefined)
target_link_libraries(target_resource_test PRIVATE "${KFX_GPU_LIBRARY}")
if(APPLE)
    target_link_libraries(target_resource_test PRIVATE "-framework Metal" "-framework QuartzCore" "-framework Foundation" "-framework AppKit" "-framework IOKit" "-framework CoreGraphics")
else()
    target_link_libraries(target_resource_test PRIVATE dl pthread m)
endif()
add_test(NAME target_resource COMMAND target_resource_test)
