cmake_minimum_required(VERSION 3.20)
project(trig_fixture C)
set(CMAKE_C_STANDARD 11)
set(KFX_SDL3_INCLUDE_DIR "" CACHE PATH "SDL3 header directory")
if(KFX_SDL3_INCLUDE_DIR)
    set(SDL3_INCLUDE_DIRS "${KFX_SDL3_INCLUDE_DIR}")
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL3 REQUIRED sdl3)
endif()
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ver_defs.h" "#define VER_STRING \"trig-test\"\n")
add_executable(trig_fixture fixture.c "${KFX_ROOT}/src/kfx/renderer/GpolyCapture.c")
target_include_directories(trig_fixture PRIVATE "${KFX_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}" ${SDL3_INCLUDE_DIRS})
target_compile_definitions(trig_fixture PRIVATE BFDEBUG_LEVEL=0)
target_compile_options(trig_fixture PRIVATE -fwrapv -fsanitize=address -fno-omit-frame-pointer)
target_link_options(trig_fixture PRIVATE -fsanitize=address)
enable_testing()
add_test(NAME trig_fixture COMMAND trig_fixture "${CMAKE_CURRENT_BINARY_DIR}/trig.bin")
add_executable(trig_determinism determinism.c "${KFX_ROOT}/src/kfx/renderer/GpolyCapture.c")
target_include_directories(trig_determinism PRIVATE "${KFX_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}" ${SDL3_INCLUDE_DIRS})
target_compile_definitions(trig_determinism PRIVATE BFDEBUG_LEVEL=0)
target_compile_options(trig_determinism PRIVATE -fwrapv -fsanitize=address -fno-omit-frame-pointer)
target_link_options(trig_determinism PRIVATE -fsanitize=address)
add_test(NAME trig_determinism COMMAND trig_determinism)
set(KFX_GPU_LIBRARY "" CACHE FILEPATH "wgpu static library")
if(KFX_GPU_LIBRARY)
    enable_language(CXX)
    set(CMAKE_CXX_STANDARD 20)
    add_executable(trig_native_bridge native.c bridge.cpp
        "${KFX_ROOT}/src/kfx/renderer/WgpuTerrainBridge.cpp" "${KFX_ROOT}/src/kfx/renderer/GpolyCapture.c")
    target_include_directories(trig_native_bridge PRIVATE "${KFX_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}" ${SDL3_INCLUDE_DIRS})
    target_compile_definitions(trig_native_bridge PRIVATE KFX_RUST_PRESENTER=1 BFDEBUG_LEVEL=0)
    target_compile_options(trig_native_bridge PRIVATE -fwrapv -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(trig_native_bridge PRIVATE -fsanitize=address)
    target_link_libraries(trig_native_bridge PRIVATE "${KFX_GPU_LIBRARY}" "-framework Metal" "-framework QuartzCore" "-framework Foundation" "-framework AppKit" "-framework IOKit" "-framework CoreGraphics")
    add_test(NAME trig_native_bridge COMMAND trig_native_bridge)
endif()
