cmake_minimum_required(VERSION 3.20)
project(primitive_fixture C)
set(CMAKE_C_STANDARD 11)
enable_testing()
set(KFX_SDL3_INCLUDE_DIR "" CACHE PATH "SDL3 headers")
if(NOT KFX_SDL3_INCLUDE_DIR)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL3 REQUIRED sdl3)
    set(KFX_SDL3_INCLUDE_DIR ${SDL3_INCLUDE_DIRS})
endif()
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ver_defs.h" "#define VER_STRING \"primitive-test\"\n")
add_executable(primitive_fixture fixture.c "${KFX_ROOT}/src/kfx/renderer/software/SwDrawTarget.c")
target_include_directories(primitive_fixture PRIVATE "${KFX_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}" ${KFX_SDL3_INCLUDE_DIR})
target_compile_definitions(primitive_fixture PRIVATE BFDEBUG_LEVEL=0)
target_compile_options(primitive_fixture PRIVATE -fwrapv -ffunction-sections -fdata-sections)
if(APPLE)
    target_link_options(primitive_fixture PRIVATE -Wl,-dead_strip)
else()
    target_link_options(primitive_fixture PRIVATE -Wl,--gc-sections)
endif()
option(KFX_PRIMITIVE_ASAN "Enable address sanitizer" OFF)
if(KFX_PRIMITIVE_ASAN)
    target_compile_options(primitive_fixture PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(primitive_fixture PRIVATE -fsanitize=address)
endif()
add_test(NAME primitive_fixture COMMAND primitive_fixture "${CMAKE_CURRENT_BINARY_DIR}/primitives.bin")
