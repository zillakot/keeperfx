cmake_minimum_required(VERSION 3.20)
project(api_socket_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
file(READ "${KFX_ROOT}/src/api.c" SOURCE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${KFX_ROOT}/src/api.c")
function(extract NAME FIRST LAST)
    string(FIND "${SOURCE}" "${FIRST}" START)
    string(FIND "${SOURCE}" "${LAST}" STOP)
    if(START LESS 0 OR STOP LESS START)
        message(FATAL_ERROR "API ${NAME} functions not found")
    endif()
    math(EXPR LENGTH "${STOP} - ${START}")
    string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} BODY)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${NAME}.inc" "${BODY}")
endfunction()
extract(platform "#ifdef _WIN32" "#include \"api.h\"")
extract(client "#define API_SERVER_BUFFER" "/**\n * Structure representing a subscribed variable.")
extract(send "static void api_send(" "/**\n * Initialize the TCP API server.")
extract(update "void api_update_server()" "/**\n * Close the API server.")
add_executable(api_socket_test socket_test.c)
target_include_directories(api_socket_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(api_socket_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME api_sockets COMMAND api_socket_test)
set_tests_properties(api_sockets PROPERTIES TIMEOUT 10)
