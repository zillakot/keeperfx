cmake_minimum_required(VERSION 3.20)
project(movie_frame_fixture CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
add_executable(movie_frame_fixture fixture.cpp "${KFX_ROOT}/src/kfx/renderer/software/MovieFrame.cpp")
target_include_directories(movie_frame_fixture PRIVATE "${KFX_ROOT}/src")
target_compile_options(movie_frame_fixture PRIVATE -fwrapv -Wall -Wextra -Werror)
option(KFX_MOVIE_ASAN "Enable address sanitizer" ON)
if(KFX_MOVIE_ASAN)
    target_compile_options(movie_frame_fixture PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(movie_frame_fixture PRIVATE -fsanitize=address)
endif()
add_test(NAME movie_frame_fixture COMMAND movie_frame_fixture "${CMAKE_CURRENT_BINARY_DIR}/movie-frames.bin")
add_executable(movie_frame_disabled disabled.cpp "${KFX_ROOT}/src/kfx/renderer/software/MovieFrame.cpp"
    "${KFX_ROOT}/src/kfx/renderer/WgpuTerrainBridge.cpp")
target_include_directories(movie_frame_disabled PRIVATE "${KFX_ROOT}/src")
target_compile_options(movie_frame_disabled PRIVATE -Wall -Wextra -Werror)
add_test(NAME movie_frame_disabled COMMAND movie_frame_disabled)
