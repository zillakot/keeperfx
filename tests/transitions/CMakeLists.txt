cmake_minimum_required(VERSION 3.20)
project(transition_fixture C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${KFX_ROOT}/src/engine_redraw.c")
file(READ "${KFX_ROOT}/src/engine_redraw.c" SOURCE)
function(extract BEGIN END OUTPUT)
    string(FIND "${SOURCE}" "${BEGIN}" START)
    string(SUBSTRING "${SOURCE}" ${START} -1 TAIL)
    string(FIND "${TAIL}" "${END}" STOP)
    if(START LESS 0 OR STOP LESS 0)
        message(FATAL_ERROR "transition extraction boundary missing")
    endif()
    string(SUBSTRING "${TAIL}" 0 ${STOP} BODY)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${OUTPUT}" "${BODY}")
endfunction()
extract("static void map_fade_native(" "void generate_map_fade_ghost_table(" fade.inc)
extract("static uint64_t capture_map_fade_buffer(" "long map_fade_in(" capture.inc)
extract("static void smooth_screen_area_native(" "void redraw_isometric_view(void)" smooth.inc)

target_compile_options(transition_fixture PRIVATE -Wall -Wextra -Werror -fwrapv -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(transition_fixture PRIVATE -fsanitize=address,undefined)
