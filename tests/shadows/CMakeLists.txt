cmake_minimum_required(VERSION 3.20)
project(shadow_fixture C)
set(CMAKE_C_STANDARD 11)
set(KFX_SDL3_INCLUDE_DIR "" CACHE PATH "SDL3 header directory")
if(KFX_SDL3_INCLUDE_DIR)
    set(SDL3_INCLUDE_DIRS "${KFX_SDL3_INCLUDE_DIR}")
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL3 REQUIRED sdl3)
endif()
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ver_defs.h" "#define VER_STRING \"shadow-test\"\n")
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/ver_defs.h"
    "#define VER_MAJOR 0\n#define VER_MINOR 0\n#define VER_RELEASE 0\n#define VER_BUILD 0\n")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/SDL3_mixer")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/SDL3_mixer/SDL_mixer.h" "#pragma once\n")
file(READ "${KFX_ROOT}/src/engine_render.c" ENGINE)
string(FIND "${ENGINE}" "static void sprite_to_sbuff(const unsigned char *sprdata" START)
string(FIND "${ENGINE}" "static void update_frontview_pointed_block(" END)
math(EXPR LENGTH "${END} - ${START}")
string(SUBSTRING "${ENGINE}" ${START} ${LENGTH} ORACLE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shadow_oracle.inc" "${ORACLE}")
add_executable(shadow_fixture fixture.c "${KFX_ROOT}/src/kfx/renderer/GpolyCapture.c")
target_include_directories(shadow_fixture PRIVATE "${KFX_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}" ${SDL3_INCLUDE_DIRS})
target_compile_definitions(shadow_fixture PRIVATE BFDEBUG_LEVEL=0)
target_compile_options(shadow_fixture PRIVATE -fwrapv -fsanitize=address -fno-omit-frame-pointer)
target_link_options(shadow_fixture PRIVATE -fsanitize=address)
enable_testing()
add_test(NAME shadow_fixture COMMAND shadow_fixture "${CMAKE_CURRENT_BINARY_DIR}/shadows.bin")
set(KFX_GPU_LIBRARY "" CACHE FILEPATH "wgpu static library")
if(KFX_GPU_LIBRARY)
    enable_language(CXX)
    set(CMAKE_CXX_STANDARD 20)
    add_executable(shadow_native fixture.c bridge.cpp "${KFX_ROOT}/src/kfx/renderer/WgpuTerrainBridge.cpp" "${KFX_ROOT}/src/kfx/renderer/GpolyCapture.c")
    target_include_directories(shadow_native PRIVATE "${KFX_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}" ${SDL3_INCLUDE_DIRS})
    target_compile_definitions(shadow_native PRIVATE KFX_RUST_PRESENTER=1 KFX_SHADOW_NATIVE=1 BFDEBUG_LEVEL=0)
    target_compile_options(shadow_native PRIVATE -fwrapv -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(shadow_native PRIVATE -fsanitize=address)
    if(APPLE)
        target_link_libraries(shadow_native PRIVATE "${KFX_GPU_LIBRARY}" "-framework Metal" "-framework QuartzCore" "-framework Foundation" "-framework AppKit" "-framework IOKit" "-framework CoreGraphics")
    else()
        target_link_libraries(shadow_native PRIVATE "${KFX_GPU_LIBRARY}" dl pthread m)
    endif()
    add_test(NAME shadow_native COMMAND shadow_native)
endif()
