cmake_minimum_required(VERSION 3.20)
project(sprite_fixture C)
set(CMAKE_C_STANDARD 11)
enable_testing()
set(KFX_SDL3_INCLUDE_DIR "" CACHE PATH "SDL3 headers")
if(NOT KFX_SDL3_INCLUDE_DIR)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL3 REQUIRED sdl3)
    set(KFX_SDL3_INCLUDE_DIR ${SDL3_INCLUDE_DIRS})
endif()
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ver_defs.h" "#define VER_STRING \"sprite-test\"\n")
set(SPRITE_SOURCES "${KFX_ROOT}/src/kfx/renderer/GpolyCapture.c"
    "${KFX_ROOT}/src/kfx/renderer/software/SwDrawTarget.c"
    "${KFX_ROOT}/src/kfx/renderer/software/WgpuSprite.c"
    "${KFX_ROOT}/src/kfx/renderer/software/bflib_vidraw_spr_norm.c"
    "${KFX_ROOT}/src/kfx/renderer/software/bflib_vidraw_spr_remp.c"
    "${KFX_ROOT}/src/kfx/renderer/software/bflib_vidraw_spr_onec.c")
option(KFX_SPRITE_ASAN "Enable address sanitizer" OFF)
foreach(fixture IN ITEMS sprite_fixture sprite_copy_fixture sprite_cache_test)
    if(fixture STREQUAL "sprite_fixture")
        set(fixture_source fixture.c)
        set(fixture_output sprites.bin)
    elseif(fixture STREQUAL "sprite_copy_fixture")
        set(fixture_source copy_fixture.c)
        set(fixture_output sprite-copies.bin)
    else()
        set(fixture_source cache_test.c)
        set(fixture_output "")
    endif()
    add_executable(${fixture} ${fixture_source} ${SPRITE_SOURCES})
    target_include_directories(${fixture} PRIVATE "${KFX_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}" ${KFX_SDL3_INCLUDE_DIR})
    target_compile_definitions(${fixture} PRIVATE BFDEBUG_LEVEL=0)
    target_compile_options(${fixture} PRIVATE -fwrapv -ffunction-sections -fdata-sections)
    if(APPLE)
        target_link_options(${fixture} PRIVATE -Wl,-dead_strip)
    else()
        target_link_options(${fixture} PRIVATE -Wl,--gc-sections)
    endif()
    if(KFX_SPRITE_ASAN)
        target_compile_options(${fixture} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
        target_link_options(${fixture} PRIVATE -fsanitize=address)
    endif()
    if(fixture_output STREQUAL "")
        add_test(NAME ${fixture} COMMAND ${fixture})
    else()
        add_test(NAME ${fixture} COMMAND ${fixture} "${CMAKE_CURRENT_BINARY_DIR}/${fixture_output}")
    endif()
endforeach()
