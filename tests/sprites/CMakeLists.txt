cmake_minimum_required(VERSION 3.20)
project(sprite_fixture C)
set(CMAKE_C_STANDARD 11)
enable_testing()
set(KFX_SDL3_INCLUDE_DIR "" CACHE PATH "SDL3 headers")
if(NOT KFX_SDL3_INCLUDE_DIR)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL3 REQUIRED sdl3)
    set(KFX_SDL3_INCLUDE_DIR ${SDL3_INCLUDE_DIRS})
endif()
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ver_defs.h" "#define VER_STRING \"sprite-test\"\n")
add_executable(sprite_fixture fixture.c "${KFX_ROOT}/src/kfx/renderer/software/SwDrawTarget.c"
    "${KFX_ROOT}/src/kfx/renderer/software/WgpuSprite.c"
    "${KFX_ROOT}/src/kfx/renderer/software/bflib_vidraw_spr_norm.c"
    "${KFX_ROOT}/src/kfx/renderer/software/bflib_vidraw_spr_remp.c"
    "${KFX_ROOT}/src/kfx/renderer/software/bflib_vidraw_spr_onec.c")
target_include_directories(sprite_fixture PRIVATE "${KFX_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}" ${KFX_SDL3_INCLUDE_DIR})
target_compile_definitions(sprite_fixture PRIVATE BFDEBUG_LEVEL=0)
target_compile_options(sprite_fixture PRIVATE -fwrapv -ffunction-sections -fdata-sections)
if(APPLE)
    target_link_options(sprite_fixture PRIVATE -Wl,-dead_strip)
else()
    target_link_options(sprite_fixture PRIVATE -Wl,--gc-sections)
endif()
option(KFX_SPRITE_ASAN "Enable address sanitizer" OFF)
if(KFX_SPRITE_ASAN)
    target_compile_options(sprite_fixture PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(sprite_fixture PRIVATE -fsanitize=address)
endif()
add_test(NAME sprite_fixture COMMAND sprite_fixture "${CMAKE_CURRENT_BINARY_DIR}/sprites.bin")
