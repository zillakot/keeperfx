cmake_minimum_required(VERSION 3.20)
project(raw_image_fixture C)
set(CMAKE_C_STANDARD 11)
enable_testing()
set(KFX_SDL3_INCLUDE_DIR "" CACHE PATH "SDL3 headers")
if(NOT KFX_SDL3_INCLUDE_DIR)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL3 REQUIRED sdl3)
    set(KFX_SDL3_INCLUDE_DIR ${SDL3_INCLUDE_DIRS})
endif()
get_filename_component(KFX_ROOT "../.." ABSOLUTE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ver_defs.h" "#define VER_STRING \"raw-image-test\"\n")
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/ver_defs.h"
    "#define VER_MAJOR 0\n#define VER_MINOR 0\n#define VER_RELEASE 0\n#define VER_BUILD 0\n")
# These drawing fixtures include sounds.h but use no SDL_mixer declarations.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/SDL3_mixer")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/SDL3_mixer/SDL_mixer.h" "#pragma once\n")
add_executable(raw_image_fixture fixture.c "${KFX_ROOT}/src/front_simple.c"
    "${KFX_ROOT}/src/gui_draw.c" "${KFX_ROOT}/src/kfx/renderer/software/WgpuRawImage.c")
target_include_directories(raw_image_fixture PRIVATE "${KFX_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}" ${KFX_SDL3_INCLUDE_DIR})
target_compile_definitions(raw_image_fixture PRIVATE BFDEBUG_LEVEL=0)
target_compile_options(raw_image_fixture PRIVATE -fwrapv -ffunction-sections -fdata-sections)
if(APPLE)
    target_link_options(raw_image_fixture PRIVATE -Wl,-dead_strip)
else()
    target_link_options(raw_image_fixture PRIVATE -Wl,--gc-sections)
endif()
option(KFX_RAW_IMAGE_ASAN "Enable address sanitizer" ON)
if(KFX_RAW_IMAGE_ASAN)
    target_compile_options(raw_image_fixture PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(raw_image_fixture PRIVATE -fsanitize=address)
endif()
add_test(NAME raw_image_fixture COMMAND raw_image_fixture "${CMAKE_CURRENT_BINARY_DIR}/raw-images.bin")
