get_filename_component(KFX_ROOT "../.." ABSOLUTE)
function(extract FILE BEGIN END OUTPUT)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${KFX_ROOT}/${FILE}")
    file(READ "${KFX_ROOT}/${FILE}" SOURCE)
    string(FIND "${SOURCE}" "${BEGIN}" START)
    string(FIND "${SOURCE}" "${END}" STOP)
    if(START LESS 0 OR STOP LESS 0)
        message(FATAL_ERROR "map fixture extraction boundary missing: ${BEGIN}")
    endif()
    math(EXPR LENGTH "${STOP}-${START}")
    string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} BODY)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${OUTPUT}" "${BODY}")
endfunction()
